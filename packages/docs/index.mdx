# kamod State

Tiny, typed reducer state management built for Preact.

```ts
import { createAction, createStore, createStoreContext } from '@kamod-ch/state'

const increment = createAction('counter/increment')
type State = { count: number }
type Action = ReturnType<typeof increment>

const store = createStore<State, Action>({
  reducer: (state = { count: 0 }, action) => increment.match(action) ? { count: state.count + 1 } : state,
})

const CounterState = createStoreContext<State, Action>({ name: 'CounterState' })
```
